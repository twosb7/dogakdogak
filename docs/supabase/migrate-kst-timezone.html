Supabase SQL 마이그레이션: UTC → Asia/Seoul (KST)

Supabase SQL Editor에서 아래 쿼리를 순서대로 실행하세요. 각 블록의 [복사] 버튼 또는 하단의 [전체 복사] 사용.

⚠ 주의: 이 마이그레이션은 기존 데이터에 영향을 주지 않습니다. 기존 행의 click_date는 그대로 유지되며, 앞으로 INSERT되는 행과 랭킹 조회 쿼리만 KST 기준으로 변경됩니다. 글로벌 함수(STEP 3~6)는 Supabase에 이미 배포된 함수를 덮어씁니다 — 실행 전 기존 함수 정의를 확인하세요.

STEP 1 — 테이블 DEFAULT 변경

ALTER TABLE
app_clicks_daily 테이블의 click_date 기본값을 KST로 변경
ALTER TABLE app_clicks_daily
  ALTER COLUMN click_date
  SET DEFAULT (CURRENT_TIMESTAMP AT TIME ZONE 'Asia/Seoul')::date;

STEP 2 — 앱별 upsert 함수 (Score + Touch)

APP FUNCTION
upsert_app_daily_clicks, upsert_app_daily_touches — INSERT 시 날짜를 KST로
CREATE OR REPLACE FUNCTION upsert_app_daily_clicks(p_data jsonb)
RETURNS void
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
BEGIN
    INSERT INTO app_clicks_daily (user_id, package_name, click_date, click_count)
    SELECT auth.uid(), d->>'package_name',
           (CURRENT_TIMESTAMP AT TIME ZONE 'Asia/Seoul')::date,
           (d->>'click_count')::bigint
    FROM jsonb_array_elements(p_data) d
    ON CONFLICT (user_id, package_name, click_date)
    DO UPDATE SET click_count = EXCLUDED.click_count;
END;
$$;

STEP 3 — 앱별 랭킹 조회 함수 (Score + Touch)

APP FUNCTION
get_app_ranking, get_app_touch_ranking — daily/weekly/monthly 필터 날짜를 KST로
CREATE OR REPLACE FUNCTION get_app_ranking(
    p_package_name text,
    p_period text DEFAULT 'alltime',
    p_limit int DEFAULT 50
)
RETURNS TABLE(
    rank bigint, user_id uuid, display_name text, avatar_url text, click_count bigint
)
LANGUAGE plpgsql STABLE SECURITY DEFINER
AS $$
DECLARE
    v_start_date date;
BEGIN
    v_start_date := CASE p_period
        WHEN 'daily'   THEN (CURRENT_TIMESTAMP AT TIME ZONE 'Asia/Seoul')::date
        WHEN 'weekly'  THEN (CURRENT_TIMESTAMP AT TIME ZONE 'Asia/Seoul')::date - INTERVAL '7 days'
        WHEN 'monthly' THEN (CURRENT_TIMESTAMP AT TIME ZONE 'Asia/Seoul')::date - INTERVAL '30 days'
        ELSE NULL
    END;

    RETURN QUERY
    SELECT
        ROW_NUMBER() OVER (ORDER BY SUM(a.click_count) DESC) AS rank,
        a.user_id,
        COALESCE(p.display_name, '익명') AS display_name,
        p.avatar_url,
        SUM(a.click_count)::bigint AS click_count
    FROM app_clicks_daily a
    JOIN profiles p ON p.id = a.user_id
    WHERE a.package_name = p_package_name
      AND a.click_count > 0
      AND (v_start_date IS NULL OR a.click_date >= v_start_date)
    GROUP BY a.user_id, p.display_name, p.avatar_url
    ORDER BY click_count DESC
    LIMIT p_limit;
END;
$$;

STEP 4 — 글로벌 upsert 함수 (Score + Touch)

GLOBAL FUNCTION
upsert_daily_clicks, upsert_daily_touches — 글로벌 일별 동기화 함수. ⚠ 실행 전 Supabase에서 기존 함수 정의를 확인하세요. 이 함수들은 user_clicks 테이블을 사용한다고 가정합니다.
-- ⚠ 실행 전 확인: 아래 명령으로 기존 함수 정의를 먼저 확인하세요
-- SELECT prosrc FROM pg_proc WHERE proname = 'upsert_daily_clicks';
-- SELECT prosrc FROM pg_proc WHERE proname = 'upsert_daily_touches';
--
-- 기존 함수에서 CURRENT_DATE 부분만 아래로 교체하면 됩니다:
--   CURRENT_DATE  →  (CURRENT_TIMESTAMP AT TIME ZONE 'Asia/Seoul')::date

-- 아래는 추정한 함수 구조입니다. 기존과 다르면 기존 함수에서 CURRENT_DATE만 바꾸세요.

CREATE OR REPLACE FUNCTION upsert_daily_clicks(p_click_count bigint)
RETURNS void
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
BEGIN
    INSERT INTO user_clicks (user_id, click_date, click_count)
    VALUES (auth.uid(), (CURRENT_TIMESTAMP AT TIME ZONE 'Asia/Seoul')::date, p_click_count)
    ON CONFLICT (user_id, click_date)
    DO UPDATE SET click_count = EXCLUDED.click_count;
END;
$$;

STEP 6 — 검증 쿼리

VERIFY
마이그레이션 후 KST가 제대로 적용됐는지 확인
-- 1. 현재 서버 시간 vs KST 시간 비교
SELECT
  now() AS server_utc,
  (now() AT TIME ZONE 'Asia/Seoul') AS kst_time,
  CURRENT_DATE AS utc_date,
  (CURRENT_TIMESTAMP AT TIME ZONE 'Asia/Seoul')::date AS kst_date;

-- 2. 함수 정의에서 Asia/Seoul 포함 여부 확인
SELECT proname, prosrc LIKE '%Asia/Seoul%' AS has_kst
FROM pg_proc
WHERE proname IN (
  'upsert_daily_clicks', 'upsert_daily_touches',
  'upsert_app_daily_clicks', 'upsert_app_daily_touches',
  'get_ranking', 'get_touch_ranking',
  'get_app_ranking', 'get_app_touch_ranking'
);
참고: STEP 4, 5의 글로벌 함수는 로컬에 원본 SQL이 없어 클라이언트 코드 기반으로 추정한 구조입니다. 실행 전 반드시 SELECT prosrc FROM pg_proc WHERE proname = '함수명';으로 기존 정의를 확인하고, 구조가 다르면 기존 함수에서 CURRENT_DATE → (CURRENT_TIMESTAMP AT TIME ZONE 'Asia/Seoul')::date만 치환하세요.
